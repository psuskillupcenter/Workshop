        // Get current filtered data
        function getFilteredData() {
            return dataset.filter(item => {
                const matchesGroup = currentFilterGroup === "ทั้งหมด" || item.groups.includes(currentFilterGroup);
                const matchesSearch = item.name.toLowerCase().includes(searchQuery) || 
                                      item.desc.toLowerCase().includes(searchQuery) ||
                                      item.groups.some(g => g.toLowerCase().includes(searchQuery));
                return matchesGroup && matchesSearch;
            });
        }

            // Update Counts
            document.getElementById('totalCount').innerText = filtered.length;
            document.getElementById('highImpactCount').innerText = filtered.filter(i => i.quadrant.startsWith('high')).length;
            document.getElementById('frontlineCount').innerText = filtered.filter(i => i.quadrant.endsWith('front')).length;
            document.getElementById('consensusCount').innerText = filtered.filter(i => i.consensus).length;

            renderMatrixDots(filtered);
            renderConsensusRankings(filtered);
            renderDirectoryTable();
        }

        // Render coordinates on 2D space
        function renderMatrixDots(items) {
            const container = document.getElementById('dotPlotContainer');
            container.innerHTML = '';

            items.forEach(item => {
                const dot = document.createElement('button');
                const styles = quadrantStyles[item.quadrant];
                
                // Highlight dot if selected
                const isSelected = item.id === selectedItemId;
                const ringClass = isSelected 
                    ? 'ring-4 ring-offset-2 ring-indigo-600 scale-125 z-30' 
                    : 'hover:scale-110 z-10';

                // Size depending on consensus rate
                const sizeClass = item.consensus ? 'w-5 h-5' : 'w-4 h-4';

                dot.className = `absolute rounded-full ${styles.dot} ${sizeClass} ${ringClass} cursor-pointer transition-all duration-150 flex items-center justify-center border-2 border-white text-[8px] text-white font-bold shadow-md`;
                dot.style.left = `${item.x}%`;
                dot.style.top = `${item.y}%`;
                dot.title = item.name;
                
                // Add inner icon or dot index indicator
                dot.innerText = item.consensus ? "★" : "";

                dot.onclick = () => selectItem(item.id);
                container.appendChild(dot);
            });
        }

        // Render consensus ranked list on right panel
        function renderConsensusRankings(filteredItems) {
            const listContainer = document.getElementById('consensusList');
            listContainer.innerHTML = '';

            // Get items sorted by consensus groups length
            const sortedConsensus = originalDataset
                .filter(i => i.groups.length > 1)
                .sort((a,b) => b.groups.length - a.groups.length);

            if (sortedConsensus.length === 0) {
                listContainer.innerHTML = '<p class="text-xs text-slate-400 py-4 text-center">ไม่มีรายการสอดคล้องกันตามเงื่อนไข</p>';
                return;
            }

            sortedConsensus.forEach(item => {
                const isSelected = item.id === selectedItemId;
                const activeStyle = isSelected ? 'bg-indigo-50 border-l-4 border-indigo-500' : 'hover:bg-slate-50';
                
                const div = document.createElement('div');
                div.className = `p-3 transition cursor-pointer flex items-center justify-between ${activeStyle}`;
                div.onclick = () => selectItem(item.id);

                div.innerHTML = `
                    <div class="space-y-0.5 pr-2">
                        <p class="font-semibold text-xs text-slate-800">${item.name}</p>
                        <p class="text-[10px] text-slate-500">เสนอโดย: ${item.groups.join(', ')}</p>
                    </div>
                    <div class="flex items-center gap-1">
                        <span class="text-[10px] font-bold bg-amber-100 text-amber-800 px-2 py-0.5 rounded-full flex items-center gap-1">
                            <i class="fa-solid fa-users"></i> ${item.groups.length} กลุ่ม
                        </span>
                    </div>
                `;
                listContainer.appendChild(div);
            });
        }

        // Render bottom full table directory
        function renderDirectoryTable() {
            const tableBody = document.getElementById('directoryTableBody');
            tableBody.innerHTML = '';

            let dataToRender = getFilteredData();
            if (directoryViewMode === 'high_impact') {
                dataToRender = dataToRender.filter(i => i.quadrant.startsWith('high'));
            }

            if (dataToRender.length === 0) {
                tableBody.innerHTML = `
                    <tr>
                        <td colspan="5" class="py-6 text-center text-slate-400 text-xs">
                            <i class="fa-solid fa-folder-open text-2xl mb-2 block text-slate-300"></i> ไม่พบรายชื่อเทคโนโลยีที่ค้นหา
                        </td>
                    </tr>
                `;
                return;
            }

            dataToRender.forEach(item => {
                const styles = quadrantStyles[item.quadrant];
                const isHighImpact = item.quadrant.startsWith('high');
                const isSelected = item.id === selectedItemId;
                
                const tr = document.createElement('tr');
                tr.className = `border-b border-slate-100 text-xs hover:bg-slate-50 transition ${isSelected ? 'bg-indigo-50/50' : ''}`;
                
                tr.innerHTML = `
                    <td class="py-3 font-medium text-slate-900 cursor-pointer" onclick="selectItem(${item.id})">
                        <div class="flex items-center gap-2">
                            ${item.consensus ? '<i class="fa-solid fa-star text-amber-500" title="Consensus"></i>' : '<i class="fa-solid fa-cube text-slate-400"></i>'}
                            <span>${item.name}</span>
                        </div>
                    </td>
                    <td class="py-3">
                        <span class="inline-flex items-center px-2 py-0.5 rounded-full text-[10px] font-medium ${styles.bg} ${styles.text}">
                            ${item.quadrant === 'high_back' ? 'High Impact | Back-office' : 
                              item.quadrant === 'high_front' ? 'High Impact | Frontline' : 
                              item.quadrant === 'low_back' ? 'Low Impact | Back-office' : 'Low Impact | Frontline'}
                        </span>
                    </td>
                    <td class="py-3 text-slate-500 max-w-[120px] truncate" title="${item.groups.join(', ')}">
                        ${item.groups.join(', ')}
                    </td>
                    <td class="py-3 text-center">
                        <span class="px-1.5 py-0.5 rounded text-[10px] font-bold ${isHighImpact ? 'bg-emerald-100 text-emerald-800' : 'bg-slate-100 text-slate-700'}">
                            ${isHighImpact ? 'สูง (High)' : 'ทั่วไป (Low)'}
                        </span>
                    </td>
                    <td class="py-3 text-right">
                        <button onclick="selectItem(${item.id})" class="text-indigo-600 hover:text-indigo-900 font-semibold text-[11px] hover:underline">
                            ดูวิเคราะห์ &rarr;
                        </button>
                    </td>
                `;
                tableBody.appendChild(tr);
            });
        }

        // 6. DETAILED VIEW PANEL HANDLER
        function selectItem(itemId) {
            selectedItemId = itemId;
            const item = originalDataset.find(i => i.id === itemId);
            if (!item) return;

            const styles = quadrantStyles[item.quadrant];

            // Update detailed card info
            const titleEl = document.getElementById('detailTitle');
            const descEl = document.getElementById('detailDesc');
            const quadBadge = document.getElementById('detailQuadBadge');
            const groupBadge = document.getElementById('detailGroupBadge');
            const consensusEl = document.getElementById('detailConsensus');

            titleEl.innerText = item.name;
            descEl.innerText = item.desc;
            
            quadBadge.innerText = styles.thai;
            quadBadge.className = `text-xs font-semibold px-2.5 py-0.5 rounded-full ${styles.bg} ${styles.text} border ${styles.border}`;
            
            groupBadge.innerText = `เสนอโดย: ${item.groups.join(', ')}`;

            if (item.consensus) {
                consensusEl.innerHTML = `<span class="text-amber-600 font-bold"><i class="fa-solid fa-fire"></i> ระบบประเมินร่วมกัน ${item.groups.length} กลุ่ม</span>`;
            } else {
                consensusEl.innerText = "แนวคิดนวัตกรรมเฉพาะหน่วยงาน";
            }

            // Bind similar search action button
            document.getElementById('viewAllSimilarBtn').onclick = () => {
                const keyword = item.name.split(' ')[0].split('/')[0];
                document.getElementById('searchInput').value = keyword;
                handleSearch(keyword);
            };

            // Rerender dot highlights
            const filtered = getFilteredData();
            renderMatrixDots(filtered);
            renderConsensusRankings(filtered);
            
            // Scroll a bit if on mobile to show description
            if (window.innerWidth < 1024) {
                document.getElementById('detailDisplayPanel').scrollIntoView({ behavior: 'smooth', block: 'nearest' });
            }
        }
